cmake_minimum_required(VERSION 3.22)
project(Frameforge VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
set(CPACK_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
set(CPACK_DEBIAN_PACKAGE_SECTION "video")
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Multimedia MultimediaWidgets Concurrent Test)
add_library(frameforge_core src/core/project.cpp src/media/media.cpp src/media/render.cpp)
target_include_directories(frameforge_core PUBLIC src)
target_link_libraries(frameforge_core PUBLIC Qt6::Core Qt6::Gui Qt6::Concurrent)
target_compile_options(frameforge_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(frameforge src/main.cpp src/ui/editor.cpp src/ui/timeline.cpp)
target_link_libraries(frameforge PRIVATE frameforge_core Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets)
target_compile_options(frameforge PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE frameforge_core Qt6::Test)
  add_test(NAME core COMMAND core_tests)
  add_executable(media_tests tests/media_tests.cpp)
  target_link_libraries(media_tests PRIVATE frameforge_core Qt6::Test)
  add_test(NAME media COMMAND media_tests)
  set_tests_properties(media PROPERTIES TIMEOUT 120)
  add_executable(desktop_tests tests/desktop_tests.cpp src/ui/editor.cpp src/ui/timeline.cpp)
  target_link_libraries(desktop_tests PRIVATE frameforge_core Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets Qt6::Test)
  add_test(NAME desktop COMMAND desktop_tests)
  add_test(NAME desktop_hidpi COMMAND desktop_tests)
  set_tests_properties(desktop desktop_hidpi PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_FFMPEG_DECODING_HW_DEVICE_TYPES=,;PULSE_SERVER=unix:/tmp/frameforge-no-pulse" TIMEOUT 90)
  set_property(TEST desktop_hidpi APPEND PROPERTY ENVIRONMENT "QT_SCALE_FACTOR=2")
endif()
install(TARGETS frameforge RUNTIME DESTINATION bin)
install(FILES packaging/frameforge.desktop DESTINATION share/applications)
install(FILES LICENSE README.md DESTINATION share/doc/frameforge)
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_CONTACT "Frameforge contributors")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6widgets6 (>= ${Qt6_VERSION}), libqt6multimedia6 (>= ${Qt6_VERSION}), libqt6multimediawidgets6 (>= ${Qt6_VERSION}), libqt6concurrent6 (>= ${Qt6_VERSION}), ffmpeg")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Frameforge experimental native video editor")
include(CPack)
